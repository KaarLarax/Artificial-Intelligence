% Crear un sistema difuso tipo Mamdani
fis = newfis('Diagnostico_Gripe');

% -----------------------
% Entrada 1: Temperatura corporal
% -----------------------
fis = addvar(fis, 'input', 'Temperatura_Corporal', [35 41]);

% Definir funciones de membresía para la Temperatura Corporal
fis = addmf(fis, 'input', 1, 'Baja', 'trimf', [35 35 37]);
fis = addmf(fis, 'input', 1, 'Moderada', 'trimf', [36 38 39]);
fis = addmf(fis, 'input', 1, 'Alta', 'trimf', [38 40 41]);

% -----------------------
% Entrada 2: Intensidad del dolor de cabeza
% -----------------------
fis = addvar(fis, 'input', 'Intensidad_Dolor_Cabeza', [0 10]);

% Definir funciones de membresía para el dolor de cabeza
fis = addmf(fis, 'input', 2, 'Ligero', 'trimf', [0 0 4]);
fis = addmf(fis, 'input', 2, 'Moderado', 'trimf', [3 5 7]);
fis = addmf(fis, 'input', 2, 'Fuerte', 'trimf', [6 8 10]);

% -----------------------
% Salida: Probabilidad de tener gripe
% -----------------------
fis = addvar(fis, 'output', 'Probabilidad_Gripe', [0 100]);

% Definir funciones de membresía para la probabilidad de gripe
fis = addmf(fis, 'output', 1, 'Baja', 'trimf', [0 0 40]);
fis = addmf(fis, 'output', 1, 'Media', 'trimf', [30 50 70]);
fis = addmf(fis, 'output', 1, 'Alta', 'trimf', [60 80 100]);

% -----------------------
% Reglas difusas
% -----------------------
ruleList = [
    1 1 1 1 1;  % Si Temperatura es Baja y Dolor es Ligero, entonces Probabilidad es Baja
    2 2 2 1 1;  % Si Temperatura es Moderada y Dolor es Moderado, entonces Probabilidad es Media
    3 3 3 1 1;  % Si Temperatura es Alta y Dolor es Fuerte, entonces Probabilidad es Alta
    3 1 2 1 1;  % Si Temperatura es Alta y Dolor es Ligero, entonces Probabilidad es Media
];

% Añadir las reglas al sistema difuso
fis = addrule(fis, ruleList);

% -----------------------
% Mostrar el sistema difuso
% -----------------------
figure;
subplot(3,1,1);
plotmf(fis, 'input', 1); % Funciones de membresía de la Temperatura Corporal
title('Funciones de membresía de la Temperatura Corporal');

subplot(3,1,2);
plotmf(fis, 'input', 2); % Funciones de membresía del Dolor de Cabeza
title('Funciones de membresía del Dolor de Cabeza');

subplot(3,1,3);
plotmf(fis, 'output', 1); % Funciones de membresía de la Probabilidad de Gripe
title('Funciones de membresía de la Probabilidad de Gripe');

% -----------------------
% Evaluar el sistema interactivamente
% -----------------------
ruleview(fis);

% También puedes probar evaluaciones directas
% Entrada: [Temperatura, Dolor de cabeza]
output = evalfis([39, 8], fis);
disp(['Probabilidad de gripe: ', num2str(output), '%']);
